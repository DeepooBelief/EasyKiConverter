50void WriteWorker::run() {
51 // 检查是否已被取消 (v3.0.5+)
52 if (m_isAborted.loadRelaxed()) {
53 m_status->writeSuccess = false;
54 m_status->writeMessage = "Export cancelled";
55 m_status->isCancelled = true;
56 m_status->addDebugLog(QString("WriteWorker cancelled for component: %1").arg(m_status->componentId));
57 emit writeCompleted(m_status);
58 return;
59 }
60
61 QElapsedTimer writeTimer;
62 writeTimer.start();
63
64 m_status->addDebugLog(QString("WriteWorker started for component: %1").arg(m_status->componentId));
65
66 // 初始化所有写入状态
67
68 m_status->symbolWritten = false;
69 m_status->footprintWritten = false;
70 m_status->model3DWritten = false;
71
72 if (!m_status->fetchSuccess || !m_status->processSuccess) {
73 m_status->writeDurationMs = 0;
74 m_status->writeSuccess = false;
75 m_status->writeMessage = "Skipped writing due to previous stage failure";
76 m_status->addDebugLog("Skipping write stage because fetch or process failed.");
77 emit writeCompleted(m_status);
78 return;
79 }
80
81 if ((m_exportSymbol && (!m_status->symbolData || m_status->symbolData->info().name.isEmpty())) &&
82 (m_exportFootprint && (!m_status->footprintData || m_status->footprintData->info().name.isEmpty()))) {
83 m_status->writeDurationMs = writeTimer.elapsed();
84 m_status->writeSuccess = false;
85 m_status->writeMessage = "No valid symbol or footprint data to write";
86 m_status->addDebugLog("ERROR: Symbol and Footprint data are empty or invalid.");
87 emit writeCompleted(m_status);
88 return;
89 }
90
91 if (!createOutputDirectory(m_outputPath)) {
92 m_status->writeDurationMs = writeTimer.elapsed();
93 m_status->writeSuccess = false;
94 m_status->writeMessage = "Failed to create output directory";
95 m_status->addDebugLog(
96 QString("ERROR: Failed to create output directory, Duration: %1ms").arg(m_status->writeDurationMs));
97 emit writeCompleted(m_status);
98 return;
99 }
100
101 // 串行写入文件，每个函数独立更新自己的状态
102 if (m_exportSymbol && m_status->symbolData) {
103 writeSymbolFile(*m_status);
104 if (m_isAborted.loadRelaxed())
105 goto cleanup;
106 }
107
108 if (m_exportFootprint && m_status->footprintData) {
109 writeFootprintFile(*m_status);
110 if (m_isAborted.loadRelaxed())
111 goto cleanup;
112 }
113
114 if (m_exportModel3D && m_status->model3DData && !m_status->model3DData->rawObj().isEmpty()) {
115 write3DModelFile(*m_status);
116 }
117
118cleanup:
119 if (m_isAborted.loadRelaxed()) {
120 m_status->writeDurationMs = writeTimer.elapsed();
121 m_status->writeSuccess = false;
122 m_status->writeMessage = "Export cancelled";
123 m_status->isCancelled = true;
124 // 状态已在开头初始化为 false，无需再次设置
125 m_status->addDebugLog(QString("WriteWorker cancelled for component: %1").arg(m_status->componentId));
126 emit writeCompleted(m_status);
127 return;
128 }
129
130 // 根据用户请求的导出项和实际完成情况，决定最终的成功状态
131 // 注意：3D模型导出失败不影响符号和封装的导出成功状态
132 // 符号和封装是捆绑的核心输出，3D模型是可选附加输出
133 bool coreSuccess = true;
134 bool model3DFailed = false;
135 if (m_exportSymbol && !m_status->symbolWritten) {
136 coreSuccess = false;
137 }
138 if (m_exportFootprint && !m_status->footprintWritten) {
139 coreSuccess = false;
140 }
141 if (m_exportModel3D && !m_status->model3DWritten) {
142 model3DFailed = true; // 仅记录，不影响核心成功状态
143 m_status->addDebugLog("WARNING: 3D model export failed, but symbol/footprint export is not affected.");
144 }
145
146 m_status->writeSuccess = coreSuccess;
147 if (coreSuccess && model3DFailed) {
148 m_status->writeMessage = "Write completed (3D model export failed, symbol/footprint OK)";
149 } else if (coreSuccess) {
150 m_status->writeMessage = "Write completed successfully";
151 } else {
152 m_status->writeMessage = "Failed to write one or more core files (symbol/footprint)";
153 }
154
155 if (m_debugMode) {
156 exportDebugData(*m_status);
157 }
158
159 m_status->writeDurationMs = writeTimer.elapsed();
160 m_status->addDebugLog(QString("WriteWorker completed for component: %1, Success: %2, Duration: %3ms")
161 .arg(m_status->componentId)
162 .arg(m_status->writeSuccess)
163 .arg(m_status->writeDurationMs));
164
165 emit writeCompleted(m_status);
166}

263bool WriteWorker::write3DModelFile(ComponentExportStatus& status) {
264 if (!status.model3DData || status.model3DData->rawObj().isEmpty()) {
265 return true;
266 }
267
268 if (!QDir(m_tempDir).exists()) {
269 status.addDebugLog(QString("ERROR: Temp directory does not exist: %1").arg(m_tempDir));
270 return false;
271 }
272 if (!PathSecurity::isSafePath(m_tempDir, m_outputPath)) {
273 status.addDebugLog(
274 QString("SECURITY ERROR: Temp dir %1 is outside output path %2").arg(m_tempDir, m_outputPath));
275 return false;
276 }
277
278 QString modelsDirPath = QString("%1/%2.3dmodels").arg(m_outputPath, m_libName);
279 if (!createOutputDirectory(modelsDirPath)) {
280 status.addDebugLog(QString("ERROR: Failed to create 3D models directory: %1").arg(modelsDirPath));
281 return false;
282 }
283
284 QString footprintName = status.footprintData ? status.footprintData->info().name : status.componentId;
285 footprintName = PathSecurity::sanitizeFilename(footprintName);
286
287 bool wrlSuccess = false;
288
289 QString wrlFilePath = QString("%1/%2.wrl").arg(modelsDirPath, footprintName);
290 wrlSuccess = AtomicFileWriter::writeAtomically(
291 m_tempDir, wrlFilePath, ".wrl.tmp", [this, &status](const QString& tempPath) -> bool {
292 return m_model3DExporter.exportToWrl(*status.model3DData, tempPath);
293 });
294
295 if (wrlSuccess) {
296 status.addDebugLog(QString("3D model WRL file written atomically: %1").arg(wrlFilePath));
297 } else {
298 status.addDebugLog(QString("ERROR: Failed to write WRL file"));
299 }
300
301 if (!status.model3DStepRaw.isEmpty()) {
302 QString stepFilePath = QString("%1/%2.step").arg(modelsDirPath, footprintName);
303 bool stepWriteSuccess = AtomicFileWriter::writeAtomically(
304 m_tempDir, stepFilePath, ".step.tmp", [&status](const QString& tempPath) -> bool {
305 QFile file(tempPath);
306 if (!file.open(QIODevice::WriteOnly)) {
307 return false;
308 }
309 qint64 written = file.write(status.model3DStepRaw);
310 file.close();
311 return written == status.model3DStepRaw.size();
312 });
313
314 if (stepWriteSuccess) {
315 status.addDebugLog(QString("3D model STEP file written"));
316 } else {
317 status.addDebugLog(QString("WARNING: Failed to write STEP file: %1").arg(stepFilePath));
318 }
319 status.clearStepData();
320 }
321
322 status.model3DWritten = wrlSuccess;
323 return wrlSuccess;
