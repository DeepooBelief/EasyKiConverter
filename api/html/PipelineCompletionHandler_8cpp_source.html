69bool PipelineCompletionHandler::exportPreviewImagesFromMemory(const QList<QByteArray>& imageDataList,
70 const QString& outputPath,
71 const QString& componentName) {
72 QDir outputDir(outputPath);
73 QString imagesDir = outputDir.filePath("images");
74 if (!outputDir.exists(imagesDir)) {
75 if (!outputDir.mkpath(imagesDir)) {
76 qWarning() << "Failed to create images directory:" << imagesDir;
77 return false;
78 }
79 }
80
81 QString safeName = componentName;
82 safeName.replace(INVALID_FILENAME_CHARS, "_");
83
84 bool allSuccess = true;
85 int exportedCount = 0;
86
87 for (int i = 0; i < imageDataList.size(); ++i) {
88 if (imageDataList[i].isEmpty()) {
89 continue;
90 }
91
92 QString filename = QString("%1_%2.jpg").arg(safeName).arg(i);
93 QString filePath = QDir(imagesDir).filePath(filename);
94
95 QFile file(filePath);
96 if (file.open(QIODevice::WriteOnly)) {
97 file.write(imageDataList[i]);
98 file.close();
99 exportedCount++;
100 } else {
101 allSuccess = false;
102 }
103 }
104
105 return allSuccess;
106}
