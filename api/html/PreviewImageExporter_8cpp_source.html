26bool PreviewImageExporter::exportFromUrls(const QStringList& imageUrls,
27 const QString& outputPath,
28 const QString& componentName) {
29 if (imageUrls.isEmpty()) {
30 return true;
31 }
32
33 if (!m_networkManager) {
34 qWarning() << "PreviewImageExporter: No network manager set";
35 return false;
36 }
37
38 QString imagesDirPath = QString("%1/%2.preview").arg(outputPath, m_options.libName);
39 if (!createPreviewDirectory(imagesDirPath)) {
40 qWarning() << "Failed to create preview images directory:" << imagesDirPath;
41 return false;
42 }
43
44 bool allSuccess = true;
45 for (int i = 0; i < imageUrls.size(); ++i) {
46 QString imageUrl = imageUrls[i];
47 QString fileExtension = QFileInfo(imageUrl).suffix();
48 if (fileExtension.isEmpty()) {
49 fileExtension = "jpg";
50 }
51
52 QString fileName = generateFileName(componentName, fileExtension, i + 1, imageUrls.size());
53 QString imagePath = QString("%1/%2").arg(imagesDirPath, fileName);
54
55 if (!m_options.overwriteExistingFiles && QFile::exists(imagePath)) {
56 qWarning() << "Preview image already exists:" << imagePath;
57 continue;
58 }
59
60 QNetworkRequest request(imageUrl);
61 request.setHeader(QNetworkRequest::UserAgentHeader, "Mozilla/5.0");
62 QNetworkReply* reply = m_networkManager->get(request);
63
64 QEventLoop loop;
65 connect(reply, &QNetworkReply::finished, &loop, &QEventLoop::quit);
66 loop.exec();
67
68 if (reply->error() == QNetworkReply::NoError) {
69 QByteArray imageData = reply->readAll();
70 QFile file(imagePath);
71 if (file.open(QIODevice::WriteOnly)) {
72 file.write(imageData);
73 file.close();
74 qDebug() << "Preview image exported successfully:" << imagePath;
75 } else {
76 qWarning() << "Failed to open file for writing:" << imagePath;
77 allSuccess = false;
78 }
79 } else {
80 qWarning() << "Failed to download preview image:" << imageUrl << reply->errorString();
81 allSuccess = false;
82 }
83
84 reply->deleteLater();
85 }
86
87 return allSuccess;
88}

90bool PreviewImageExporter::exportFromCache(const QStringList& imagePaths,
91 const QString& outputPath,
92 const QString& componentName) {
93 if (imagePaths.isEmpty()) {
94 return true;
95 }
96
97 // exportFromCache 只进行文件复制操作，不需要网络管理器
98 QString imagesDirPath = QString("%1/%2.preview").arg(outputPath, m_options.libName);
99 if (!createPreviewDirectory(imagesDirPath)) {
100 qWarning() << "Failed to create preview images directory:" << imagesDirPath;
101 return false;
102 }
103
104 bool allSuccess = true;
105 for (int i = 0; i < imagePaths.size(); ++i) {
106 QString sourcePath = imagePaths[i];
107
108 if (!QFile::exists(sourcePath)) {
109 qWarning() << "Source preview image file not found:" << sourcePath;
110 allSuccess = false;
111 continue;
112 }
113
114 QString fileExtension = QFileInfo(sourcePath).suffix();
115 if (fileExtension.isEmpty()) {
116 fileExtension = "jpg";
117 }
118
119 QString fileName = generateFileName(componentName, fileExtension, i + 1, imagePaths.size());
120 QString destPath = QString("%1/%2").arg(imagesDirPath, fileName);
121
122 if (!m_options.overwriteExistingFiles && QFile::exists(destPath)) {
123 qWarning() << "Preview image already exists:" << destPath;
124 continue;
125 }
126
127 if (QFile::copy(sourcePath, destPath)) {
128 qDebug() << "Preview image copied from cache:" << destPath;
129 } else {
130 qWarning() << "Failed to copy preview image from cache:" << sourcePath << "to" << destPath;
131 allSuccess = false;
132 }
133 }
134
135 return allSuccess;
136}
