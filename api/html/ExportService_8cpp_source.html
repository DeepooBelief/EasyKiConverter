129void ExportService::executeExportPipelineWithData(const QList<ComponentData>& componentDataList,
130 const ExportOptions& options) {
131 qDebug() << "Executing export pipeline with data for" << componentDataList.size() << "components";
132
133 QMutexLocker locker(m_mutex);
134
135 if (m_isExporting) {
136 qWarning() << "Export already in progress";
137 return;
138 }
139
140 m_isExporting = true;
141 m_options = options;
142 m_currentProgress = 0;
143 m_totalProgress = componentDataList.size();
144 m_successCount = 0;
145 m_failureCount = 0;
146
147 // 清空之前的数
148 m_exportDataList.clear();
149
150 // 收集所有符号和封装数据
151 QList<SymbolData> allSymbols;
152 QList<FootprintData> allFootprints;
153 QList<Model3DData> allModels;
154
155 locker.unlock();
156
157 // 为每个元件收集数
158 for (const ComponentData& componentData : componentDataList) {
159 ExportData exportData;
160 exportData.componentId = componentData.lcscId();
161
162 // 检查并复制符号数据
163 if (componentData.symbolData() && !componentData.symbolData()->info().name.isEmpty()) {
164 exportData.symbolData = *componentData.symbolData();
165 allSymbols.append(exportData.symbolData);
166 }
167
168 // 检查并复制封装数据
169 if (componentData.footprintData() && !componentData.footprintData()->info().name.isEmpty()) {
170 exportData.footprintData = *componentData.footprintData();
171 allFootprints.append(exportData.footprintData);
172 }
173
174 // 检查并复制3D模型数据
175 if (componentData.model3DData() && !componentData.model3DData()->uuid().isEmpty()) {
176 exportData.model3DData = *componentData.model3DData();
177 allModels.append(exportData.model3DData);
178 }
179
180 // 收集预览图数据
181 if (!componentData.previewImages().isEmpty()) {
182 exportData.previewImages = componentData.previewImages();
183 }
184
185 // 收集手册数据
186 if (!componentData.datasheet().isEmpty()) {
187 exportData.datasheet = componentData.datasheet();
188 }
189
190 exportData.success = false;
191 m_exportDataList.append(exportData);
192
193 qDebug() << "Added export data for:" << exportData.componentId
194 << "Symbol:" << !exportData.symbolData.info().name.isEmpty()
195 << "Footprint:" << !exportData.footprintData.info().name.isEmpty()
196 << "3D Model:" << !exportData.model3DData.uuid().isEmpty()
197 << "Preview Images:" << exportData.previewImages.size()
198 << "Datasheet:" << !exportData.datasheet.isEmpty();
199 }
200
201 // 创建输出目录
202 if (!createOutputDirectory(m_options.outputPath)) {
203 qWarning() << "Failed to create output directory:" << m_options.outputPath;
204 m_isExporting = false;
205 emit exportFailed("Failed to create output directory");
206 return;
207 }
208
209 // 导出符号库
210 if (m_options.exportSymbol && !allSymbols.isEmpty()) {
211 QString symbolLibPath = QString("%1/%2.kicad_sym").arg(m_options.outputPath, m_options.libName);
212 if (exportSymbolLibrary(allSymbols, m_options.libName, symbolLibPath)) {
213 qDebug() << "Symbol library exported successfully:" << symbolLibPath;
214 // 在追加模式下，如果所有符号都已存在，exportSymbolLibrary 会返true
215 // 但实际上没有导出任何新符号，这种情况下我们认为这些符号已经存
216 // 不应该计入成功或失败计数
217 // 只有在实际导出新符号或覆盖现有符号时才增加计
218 // 由于我们无法从返回值判断是否实际导出了新符
219 // 这里简单处理：如果文件存在，认为所有符号都已成功导出（包括之前已存在的
220 m_successCount += allSymbols.size();
221 } else {
222 qWarning() << "Failed to export symbol library";
223 m_failureCount += allSymbols.size();
224 }
225 }
226
227 // 导出3D模型（需要在导出封装库之前完成）
228 QString modelsDirPath = QString("%1/3dmodels").arg(m_options.outputPath);
229 if (m_options.exportModel3D && !allModels.isEmpty()) {
230 // 创建 3D 模型目录（添加库名称前缀
231 QString modelsDirPath = QString("%1/%2.3dmodels").arg(m_options.outputPath, m_options.libName);
232 if (!createOutputDirectory(modelsDirPath)) {
233 qWarning() << "Failed to create 3D models directory:" << modelsDirPath;
234 } else {
235 // 为每3D 模型设置文件名（使用封装名称
236 QMap<QString, QString> modelPathMap; // UUID -> 文件路径映射
237
238 for (auto& model : allModels) {
239 // 查找对应的封装名
240 QString footprintName;
241 for (const auto& exportData : m_exportDataList) {
242 if (exportData.model3DData.uuid() == model.uuid() &&
243 !exportData.footprintData.info().name.isEmpty()) {
244 footprintName = exportData.footprintData.info().name;
245 break;
246 }
247 }
248
249 // 使用封装名称作为文件
250 QString modelName = footprintName.isEmpty() ? model.uuid() : footprintName;
251 QString wrlPath = QString("%1/%2.wrl").arg(modelsDirPath, modelName);
252 QString stepPath = QString("%1/%2.step").arg(modelsDirPath, modelName);
253
254 // 保存 WRL 模型
255 if (!model.rawObj().isEmpty()) {
256 if (!m_options.overwriteExistingFiles && QFile::exists(wrlPath)) {
257 qWarning() << "WRL model file already exists:" << wrlPath;
258 } else {
259 if (m_modelExporter->exportToWrl(model, wrlPath)) {
260 qDebug() << "WRL model exported successfully:" << wrlPath;
261 } else {
262 qWarning() << "Failed to export WRL model:" << modelName;
263 }
264 }
265 }
266
267 // 保存 STEP 模型
268 if (!model.step().isEmpty() && model.step().size() > 100) {
269 if (!m_options.overwriteExistingFiles && QFile::exists(stepPath)) {
270 qWarning() << "STEP model file already exists:" << stepPath;
271 } else {
272 if (m_modelExporter->exportToStep(model, stepPath)) {
273 qDebug() << "STEP model exported successfully:" << stepPath;
274 } else {
275 qWarning() << "Failed to export STEP model:" << modelName;
276 }
277 }
278 }
279
280 // 保存模型路径映射（使用相对路径，相对于封装库目录
281 QString relativePath = QString("${KIPRJMOD}/%1.3dmodels/%2").arg(m_options.libName, modelName);
282 modelPathMap[model.uuid()] = relativePath;
283 }
284
285 // 更新封装数据中的 3D 模型路径
286 for (auto& footprint : allFootprints) {
287 if (!footprint.model3D().uuid().isEmpty() && modelPathMap.contains(footprint.model3D().uuid())) {
288 Model3DData updatedModel = footprint.model3D();
289 updatedModel.setName(modelPathMap[footprint.model3D().uuid()]);
290 footprint.setModel3D(updatedModel);
291 }
292 }
293 }
294 }
295
296 // 导出预览图
297 qDebug() << "ExportPreviewImages option:" << m_options.exportPreviewImages;
298 qDebug() << "Total components:" << m_exportDataList.size();
299 int componentsWithPreviewImages = 0;
300 for (const auto& exportData : m_exportDataList) {
301 if (!exportData.previewImages.isEmpty()) {
302 componentsWithPreviewImages++;
303 }
304 }
305 qDebug() << "Components with preview images:" << componentsWithPreviewImages;
306
307 if (m_options.exportPreviewImages) {
308 int previewImageSuccessCount = 0;
309 for (const auto& exportData : m_exportDataList) {
310 if (!exportData.previewImages.isEmpty()) {
311 qDebug() << "Exporting preview images for:" << exportData.componentId
312 << "Image count:" << exportData.previewImages.size();
313
314 // 使用组件名称或ID作为文件名
315 QString componentName = exportData.symbolData.info().name.isEmpty() ? exportData.componentId
316 : exportData.symbolData.info().name;
317
318 if (exportPreviewImages(exportData.previewImages, m_options.outputPath, componentName)) {
319 previewImageSuccessCount++;
320 }
321 }
322 }
323 qDebug() << "Preview images export completed:" << previewImageSuccessCount << "components";
324 }
325
326 // 导出手册
327 qDebug() << "ExportDatasheet option:" << m_options.exportDatasheet;
328 int componentsWithDatasheets = 0;
329 for (const auto& exportData : m_exportDataList) {
330 if (!exportData.datasheet.isEmpty()) {
331 componentsWithDatasheets++;
332 }
333 }
334 qDebug() << "Components with datasheets:" << componentsWithDatasheets;
335
336 if (m_options.exportDatasheet) {
337 int datasheetSuccessCount = 0;
338 for (const auto& exportData : m_exportDataList) {
339 if (!exportData.datasheet.isEmpty()) {
340 qDebug() << "Exporting datasheet for:" << exportData.componentId
341 << "Datasheet URL:" << exportData.datasheet;
342
343 // 使用组件名称或ID作为文件名
344 QString componentName = exportData.symbolData.info().name.isEmpty() ? exportData.componentId
345 : exportData.symbolData.info().name;
346
347 if (exportDatasheet(exportData.datasheet, m_options.outputPath, componentName)) {
348 datasheetSuccessCount++;
349 }
350 }
351 }
352 qDebug() << "Datasheets export completed:" << datasheetSuccessCount << "components";
353 }
354
355 // 创建封装库目录
356 QString footprintDirPath = QString("%1/%2.pretty").arg(m_options.outputPath, m_options.libName);
357 if (m_options.exportFootprint && !allFootprints.isEmpty()) {
358 if (exportFootprintLibrary(allFootprints, m_options.libName, footprintDirPath)) {
359 qDebug() << "Footprint library exported successfully with 3D model paths:" << footprintDirPath;
360 } else {
361 m_failureCount += allFootprints.size();
362 }
363 }
364
365 // 更新进度和统
366 m_successCount = m_totalProgress - m_failureCount;
367 m_currentProgress = m_totalProgress;
368
369 // 发送导出完成信
370 emit exportProgress(m_currentProgress, m_totalProgress);
371
372 // 为每个元件发送导出成功信
373 for (const ExportData& exportData : m_exportDataList) {
374 emit componentExported(exportData.componentId, true, "Export successful");
375 }
376
377 // 完成导出
378 m_isExporting = false;
379 qDebug() << "Export pipeline completed:" << m_successCount << "success," << m_failureCount << "failed";
380 emit exportCompleted(m_totalProgress, m_successCount);
381}

517void ExportService::processNextExport() {
518 QMutexLocker locker(m_mutex);
519
520 if (!m_isExporting) {
521 return;
522 }
523
524 // 检查是否所有元件都已处
525 if (m_currentProgress >= m_totalProgress) {
526 m_isExporting = false;
527 locker.unlock();
528
529 qDebug() << "Export pipeline completed:" << m_successCount << "success," << m_failureCount << "failed";
530 emit exportCompleted(m_totalProgress, m_successCount);
531 return;
532 }
533
534 // 获取下一个待处理的元
535 int index = m_currentProgress;
536 if (index >= m_exportDataList.size()) {
537 return;
538 }
539
540 ExportData& exportData = m_exportDataList[index];
541 QString componentId = exportData.componentId;
542 locker.unlock();
543
544 // 检查是否有数据可以导出
545 bool hasSymbolData = !exportData.symbolData.info().name.isEmpty();
546 bool hasFootprintData = !exportData.footprintData.info().name.isEmpty();
547 bool hasModel3DData = !exportData.model3DData.uuid().isEmpty();
548
549 if (!hasSymbolData && !hasFootprintData && !hasModel3DData) {
550 qWarning() << "No data available for export:" << componentId;
551 handleExportTaskFinished(componentId, false, "No data available for export");
552 return;
553 }
554
555 // 创建输出目录
556 if (!createOutputDirectory(m_options.outputPath)) {
557 qWarning() << "Failed to create output directory:" << m_options.outputPath;
558 handleExportTaskFinished(componentId, false, "Failed to create output directory");
559 return;
560 }
561
562 // 导出符号
563 if (m_options.exportSymbol && hasSymbolData) {
564 QString symbolPath = QString("%1/%2.kicad_sym").arg(m_options.outputPath, componentId);
565 if (!exportSymbol(exportData.symbolData, symbolPath)) {
566 qWarning() << "Failed to export symbol for:" << componentId;
567 handleExportTaskFinished(componentId, false, "Failed to export symbol");
568 return;
569 }
570 }
571
572 // 导出封装
573 if (m_options.exportFootprint && hasFootprintData) {
574 QString footprintPath = QString("%1/%2.kicad_mod").arg(m_options.outputPath, componentId);
575 if (!exportFootprint(exportData.footprintData, footprintPath)) {
576 qWarning() << "Failed to export footprint for:" << componentId;
577 handleExportTaskFinished(componentId, false, "Failed to export footprint");
578 return;
579 }
580 }
581
582 // 导出3D模型
583 if (m_options.exportModel3D && hasModel3DData) {
584 QString modelPath = QString("%1/%2.wrl").arg(m_options.outputPath, exportData.model3DData.uuid());
585 if (!export3DModel(exportData.model3DData, modelPath)) {
586 qWarning() << "Failed to export 3D model for:" << componentId;
587 handleExportTaskFinished(componentId, false, "Failed to export 3D model");
588 return;
589 }
590 }
591
592 // 导出成功
593 handleExportTaskFinished(componentId, true, "Export successful");
594}

603 qWarning() << "Export already in progress";

606
607 m_isExporting = true;

652 if (exportSymbolLibrary(allSymbols, m_options.libName, symbolLibPath)) {
653 qDebug() << "Symbol library exported successfully:" << symbolLibPath;
654 // 在追加模式下，如果所有符号都已存在，exportSymbolLibrary 会返true
655 // 但实际上没有导出任何新符号，这种情况下我们认为这些符号已经存
656 // 不应该计入成功或失败计数
657 // 只有在实际导出新符号或覆盖现有符号时才增加计
658 // 由于我们无法从返回值判断是否实际导出了新符
659 // 这里简单处理：如果文件存在，认为所有符号都已成功导出（包括之前已存在的
660 m_successCount += allSymbols.size();
661 } else {
662 qWarning() << "Failed to export symbol library";
663 m_failureCount += allSymbols.size();
664 }
665 }
666
667 // 导出3D模型（需要在导出封装库之前完成）
668 QString modelsDirPath = QString("%1/%2.3dmodels").arg(m_options.outputPath, m_options.libName);
669 if (m_options.exportModel3D && !allModels.isEmpty()) {
670 // 创建 3D 模型目录
671 if (!createOutputDirectory(modelsDirPath)) {
672 qWarning() << "Failed to create 3D models directory:" << modelsDirPath;
673 } else {
674 // 为每3D 模型设置文件名（使用封装名称
675 QMap<QString, QString> modelPathMap; // UUID -> 文件路径映射
676
677 for (auto& model : allModels) {
678 // 查找对应的封装名
679 QString footprintName;
680 for (const auto& componentData : componentDataList) {
681 if (componentData.model3DData() && componentData.model3DData()->uuid() == model.uuid() &&
682 componentData.footprintData() && !componentData.footprintData()->info().name.isEmpty()) {
683 footprintName = componentData.footprintData()->info().name;
684 break;
685 }
686 }
687
688 // 使用封装名称作为文件
689 QString modelName = footprintName.isEmpty() ? model.uuid() : footprintName;
690 QString wrlPath = QString("%1/%2.wrl").arg(modelsDirPath, modelName);
691 QString stepPath = QString("%1/%2.step").arg(modelsDirPath, modelName);
692
693 // 保存 WRL 模型
694 if (!model.rawObj().isEmpty()) {
695 if (!m_options.overwriteExistingFiles && QFile::exists(wrlPath)) {
696 qWarning() << "WRL model file already exists:" << wrlPath;
697 } else {
698 if (m_modelExporter->exportToWrl(model, wrlPath)) {
699 qDebug() << "WRL model exported successfully:" << wrlPath;
700 } else {
701 qWarning() << "Failed to export WRL model:" << modelName;
702 }
703 }
704 }
705
706 // 保存 STEP 模型
707 if (!model.step().isEmpty() && model.step().size() > 100) {
708 if (!m_options.overwriteExistingFiles && QFile::exists(stepPath)) {
709 qWarning() << "STEP model file already exists:" << stepPath;
710 } else {
711 if (m_modelExporter->exportToStep(model, stepPath)) {
712 qDebug() << "STEP model exported successfully:" << stepPath;
713 } else {
714 qWarning() << "Failed to export STEP model:" << modelName;
715 }
716 }
717 }
718
719 // 保存模型路径映射（使用相对路径，相对于封装库目录
720 QString relativePath = QString("${KIPRJMOD}/%1.3dmodels/%2").arg(m_options.libName, modelName);
721 modelPathMap[model.uuid()] = relativePath;
722 }
723
724 // 更新封装数据中的 3D 模型路径
725 for (auto& footprint : allFootprints) {
726 if (!footprint.model3D().uuid().isEmpty() && modelPathMap.contains(footprint.model3D().uuid())) {
727 Model3DData updatedModel = footprint.model3D();
728 updatedModel.setName(modelPathMap[footprint.model3D().uuid()]);
729 // 保留 STEP 数据
730 for (const auto& model : allModels) {
731 if (model.uuid() == footprint.model3D().uuid()) {
732 updatedModel.setStep(model.step());
733 break;
734 }
735 }
736 footprint.setModel3D(updatedModel);
737 }
738 }
739 }
740 }
741
742 // 创建封装库目录
743 QString footprintDirPath = QString("%1/%2.pretty").arg(m_options.outputPath, m_options.libName);
744 if (m_options.exportFootprint && !allFootprints.isEmpty()) {
745 if (exportFootprintLibrary(allFootprints, m_options.libName, footprintDirPath)) {
746 qDebug() << "Footprint library exported successfully with 3D model paths:" << footprintDirPath;
747 m_successCount += allFootprints.size();
748 } else {
749 m_failureCount += allFootprints.size();
750 }
751 }
752
753 // 更新进度和统
754 m_currentProgress = m_totalProgress;
755
756 // 发送导出完成信
757 m_isExporting = false;
758 m_parallelExporting = false;
759 emit exportProgress(m_currentProgress, m_totalProgress);
760 emit exportCompleted(m_totalProgress, m_successCount);
761}

763void ExportService::handleParallelExportTaskFinished(const QString& componentId, bool success, const QString& message) {
764 qDebug() << "Parallel export task finished for:" << componentId << "Success:" << success;
765
766 QMutexLocker locker(m_mutex);
767
768 // 更新统计信息
769 if (success) {
770 m_successCount++;
771 } else {
772 m_failureCount++;
773 }
774
775 // 更新状
776 m_parallelExportStatus[componentId] = false;
777 m_parallelCompletedCount++;
778
779 // 更新进度
780 m_currentProgress++;
781 locker.unlock();
782
783 emit exportProgress(m_currentProgress, m_totalProgress);
784 emit componentExported(componentId, success, message);
785
786 // 检查是否所有任务都已完
787 if (m_parallelCompletedCount >= m_parallelTotalCount) {
788 qDebug() << "All parallel export tasks completed:" << m_successCount << "success," << m_failureCount
789 << "failed";
790
791 m_isExporting = false;
792 m_parallelExporting = false;
793 emit exportCompleted(m_totalProgress, m_successCount);
794 }
